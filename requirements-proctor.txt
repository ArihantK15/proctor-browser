# Python dependencies for proctor.py (the local face/object/voice detector
# spawned by the Electron app during an active exam). These are SEPARATE
# from requirements.txt, which only lists the FastAPI backend deps that run
# inside the Docker container on the server. proctor.py runs on the
# student's local machine alongside Electron and is not packaged in the
# server image.
#
# Install with:  pip3 install --break-system-packages -r requirements-proctor.txt
#
# Face detection: uniface (RetinaFace ONNX) — replaces mediapipe, which
# crashed on Python 3.12 + Apple Silicon with a mutex-lock abort. uniface
# pulls onnxruntime + opencv-python and gives us face boxes plus the 5
# landmarks we feed into solvePnP for head pose.
#
# Gaze: weights/resnet18_gaze.onnx is loaded directly via onnxruntime.
# Eye open/closed: OpenCV Haar cascade (ships with opencv-python).
#
# Objects (phone/book/laptop/…): YOLO runs on onnxruntime from a
# pre-exported, bundled ONNX — NOT ultralytics/torch. Preferred model is
# weights/yolo26n.onnx (YOLO26: NMS-free / end-to-end, faster CPU inference);
# the proctor falls back to legacy weights/yolov8n.onnx if it's absent, and
# _yolo_infer auto-detects the output head either way.
# Dropping ultralytics removes the ~2 GB torch dependency entirely.
# Re-export the model offline (needs ultralytics+torch on a dev box only)
# with (YOLO26 end-to-end is the default):
#     yolo export model=yolo26n.pt format=onnx imgsz=640 opset=19
# then commit the result to weights/yolo26n.onnx (shipped via the
# Electron extraResources weights/ mapping). The legacy export command was
#     yolo export model=yolov8n.pt format=onnx imgsz=640 opset=12

# PINNED: uniface's detect() return shape changed across majors — 1.1.x returns
# dicts (what detect_faces parses), 3.x returns Face OBJECTS, so face.get('bbox')
# raised "'Face' object has no attribute 'get'" on every frame and detection
# silently died. `>=1.1.0` had let the bundle pull 3.7.0. Pin uniface EXACTLY
# (its retinaface_mnet_v2.onnx is SHA-locked to it) and cap the rest below the
# next major so an API break can't slip in unnoticed. Keep in lockstep with
# config.js#PIP_PACKAGES (the bundle installs from there).
uniface==3.7.1
onnxruntime>=1.19.2,<2
opencv-python>=4.13.0.92,<5
numpy>=1.24.0,<3
insightface>=0.7.3,<0.8
sounddevice>=0.5.5
requests>=2.34.2
websocket-client>=1.9.0
# psutil — used by proctor.py's adaptive _HardwareGovernor to read
# CPU% every 5 s and back off the per-frame cadence when the student's
# laptop is heat-throttling. Soft-imported; if it's missing the
# governor permanently returns TARGET_FPS (i.e. behaves like the old
# code).
psutil>=7.2.2

# ── Phase 75 — on-device audio detection ──────────────────────────
# Soft-imported in audio_processor.py: if either dep is missing the
# proctor still runs and the existing RMS-only voice path is
# unaffected. We log "audio processor unavailable" once and skip
# the new keyword_uttered / multiple_voices_detected events.
#
# vosk          — offline ASR. Models live in ./weights/ and are
#                 downloaded out-of-band by
#                 scripts/download_audio_models.sh (~40 MB each for
#                 en-IN + hi-IN). NOT bundled in git.
# python_speech_features — MFCC extractor used by the two-cluster
#                 heuristic for "≥ 2 voices?" detection. Pure Python,
#                 ~50 KB, no native build.
vosk>=0.3.45
python_speech_features>=0.6
